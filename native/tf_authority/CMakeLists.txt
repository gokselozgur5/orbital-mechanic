cmake_minimum_required(VERSION 3.16)
project(orbital_tf_authority LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(orbital_tf_authority audit.cpp)
target_compile_features(orbital_tf_authority PRIVATE cxx_std_17)
target_compile_options(orbital_tf_authority PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(orbital_tf_authority rclcpp tf2_msgs)
target_link_libraries(orbital_tf_authority nlohmann_json::nlohmann_json)
